cmake_minimum_required(VERSION 3.16)
project(SuperJupiterSensorAgreementConsumer LANGUAGES CXX)
if(DEFINED SUPERJUPITER_SENSOR_AGREEMENT_SOURCE_DIR)
  add_subdirectory("${SUPERJUPITER_SENSOR_AGREEMENT_SOURCE_DIR}" sensor-agreement)
  if(TARGET superjupiter-sensor-agreement-tests OR TARGET superjupiter-sensor-agreement-replay)
    message(FATAL_ERROR "A consumer unexpectedly builds library developer tools")
  endif()
else()
  find_package(SuperJupiterSensorAgreement 0.1 CONFIG REQUIRED)
endif()
add_executable(superjupiter-sensor-agreement-consumer main.cpp)
target_link_libraries(superjupiter-sensor-agreement-consumer PRIVATE SuperJupiter::SensorAgreement)
